use cm_core::import::{normalize, PaperDraft};
use cm_metadata::{bibtex, ris};
use std::{collections::BTreeMap, fs, path::Path};

fn normalized(d: &PaperDraft) -> serde_json::Value {
    let d = normalize::normalize(d).unwrap();
    serde_json::json!({"title":d.title,"bibliography":d.bibliography,"external_ids":d.external_ids})
}

#[test]
fn actual_native_exports_match_the_imported_fixture() {
    let root = Path::new("/private/tmp/cm-pr45-latest-bulk-cjxuh0og");
    let source = bibtex::parse(&fs::read(root.join("inputs/thousand.bib")).unwrap()).unwrap();
    assert!(source.diagnostics.is_empty());
    assert_eq!(source.candidates.len(), 1000);
    let expected: BTreeMap<_, _> = source.candidates.iter().map(|d| {
        (normalize::normalize(d).unwrap().bibliography.doi.unwrap(), normalized(d))
    }).collect();
    assert_eq!(expected.len(), 1000);
    for (name, count) in [("all.ris",1000),("selected.ris",1),("collection.ris",1)] {
        let report = ris::parse(&fs::read(root.join(name)).unwrap()).unwrap();
        assert!(report.diagnostics.is_empty(),"{name}: {:?}",report.diagnostics);
        assert_eq!(report.candidates.len(),count,"{name}");
        let mut seen=std::collections::BTreeSet::new();
        for draft in report.candidates {
            assert!(!draft.diagnostics.iter().any(|d|d.error),"{name}: {:?}",draft.diagnostics);
            let key=normalize::normalize(&draft).unwrap().bibliography.doi.unwrap();
            assert!(seen.insert(key.clone()));
            assert_eq!(normalized(&draft),expected[&key],"{name}: {key}");
        }
        println!("{name}: {count} unique records, normalized title/bibliography/external IDs equal, no error diagnostics");
    }
}
